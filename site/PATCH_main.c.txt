#define WEB_JSON      "/tmp/triolink.json"
#define WEB_JSON_TMP  "/tmp/triolink.json.tmp"

/* Grava os dados para o site; escreve em arquivo temporário e renomeia (atômico) */
static void escrever_json_web(const SensorData *b, const SensorData *m) {
    FILE *f = fopen(WEB_JSON_TMP, "w");
    if (!f) return;
    fprintf(f,
        "{\"bmp\":{\"online\":%s,\"pressao\":%.2f,\"temperatura\":%.2f,\"altitude\":%.2f,\"estado\":\"%s\"},"
        "\"mpu\":{\"online\":%s,\"direcao\":%.2f,\"velocidade\":%.2f,\"estado\":\"%s\"}}\n",
        b->online ? "true" : "false", b->v1, b->v2, b->v3, b->estado,
        m->online ? "true" : "false", m->v1, m->v2, m->estado);
    fclose(f);
    rename(WEB_JSON_TMP, WEB_JSON);
}
